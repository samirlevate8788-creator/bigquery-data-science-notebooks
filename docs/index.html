Multimodal Vector Search Simulator

Cosine similarity lookup against BigQuery IVF vector index

SELECT base.id, base.price, base.sq_ft, distance
FROM VECTOR_SEARCH(
  TABLE `housing_dataset.home_embeddings`, 'mm_embedding',
  (ML.GENERATE_EMBEDDING(MODEL `multimodal_embedding_model`, (SELECT "house near the ocean" AS content))),
  top_k => 3
)
ORDER BY distance ASC;
